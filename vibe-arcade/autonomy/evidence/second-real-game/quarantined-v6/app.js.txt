(function(){
'use strict';

function setReduce(){reduce=!!(mq&&mq.matches);var shell=document.querySelector('[data-cargo-shell]');if(shell)shell.setAttribute('data-reduced-motion',String(reduce));}
setReduce();if(mq){if(mq.addEventListener)mq.addEventListener('change',setReduce);else mq.addListener(setReduce);}
function text(sel,v){var e=document.querySelector(sel);if(e)e.textContent=v;}

 var result=document.querySelector('[data-cargo-result]'), detail=document.querySelector('[data-cargo-result-detail]');if(state.phase==='finished'||state.phase==='incomplete'){if(result){result.hidden=false;result.textContent=state.phase==='finished'?'SKYPORT COMPLETE':'CARGO INCOMPLETE';}if(detail)detail.textContent=(state.stage||0)+'/3 dispatched · '+(state.assignments||0)+' assignments · Score '+(state.score||0)+(state.phase==='incomplete'?' — Recheck each bay’s mass and energy.':' — Fleet departure authorized.');}else if(result)result.hidden=true;
 opts=opts||{};opts.reducedMotion=reduce;opts.width=opts.width||ctx.canvas.width;opts.height=opts.height||ctx.canvas.height;window.CloudCargoPresentation.draw(ctx,cloned,opts);
}
window.CargoPresentation={snapshot:function(){return {reducedMotion:reduce,particles:0,activeVoices:0};}};

}());
