(function(){
'use strict';
function rng(seed){var x=(seed>>>0)||1;return function(){x^=x<<13;x^=x>>>17;x^=x<<5;return (x>>>0)/4294967296;};}
function shuffle(a,r){for(var i=a.length-1;i>0;i--){var j=(r()*(i+1))|0,t=a[i];a[i]=a[j];a[j]=t;}return a;}
function board(s,stage){var r=rng((s.seed+Math.imul(stage+1,2654435761))>>>0), n=4+stage, bn=stage?3:2, cards=[], groups=[];for(var b=0;b<bn;b++)groups[b]=[];
 for(var i=0;i<n;i++){var m=1+((r()*5)|0), e=stage?((r()*4)|0):0, b=i%bn;cards.push({mass:m,energy:e,bay:b});groups[b].push(cards.length-1);}
 /* Make late energy a genuine second constraint: two equal-mass cards have distinct energy. */
 if(stage===2){cards[0].mass=2;cards[0].energy=0;cards[1].mass=2;cards[1].energy=3;cards[0].bay=0;cards[1].bay=1;groups=[[],[],[]];for(i=0;i<n;i++)groups[cards[i].bay].push(i);}
 var bays=[];for(b=0;b<bn;b++){var mm=0,ee=0;for(var q=0;q<groups[b].length;q++){var c=cards[groups[b][q]];mm+=c.mass;ee+=c.energy;}bays.push({mass:mm,energy:stage?ee:null});}
 shuffle(cards,r);shuffle(bays,r); /* target order is intentionally independent; remap cards to their matching target */
 /* Reconstruct target identity by sums before shuffle would be impossible after it, so generate a permutation map. */
 /* The cards retain group labels only temporarily in this local function. */
 var targets=[];for(b=0;b<bn;b++)targets.push({mass:0,energy:stage?0:null,old:b});
 for(i=0;i<n;i++){var cc=cards[i], old=cc.bay;targets[old].mass+=cc.mass;if(stage)targets[old].energy+=cc.energy;}
 shuffle(targets,r);for(i=0;i<n;i++){old=cards[i].bay;for(b=0;b<bn;b++)if(targets[b].old===old){cards[i].bay=-1;break;}}
 for(b=0;b<bn;b++)delete targets[b].old;
 s.cards=cards;s.bays=targets;s.selectedCard=0;s.focus=0;s.validation=0;s.phase='play';
}
function totals(s,b){var m=0,e=0;for(var i=0;i<s.cards.length;i++)if(s.cards[i].bay===b){m+=s.cards[i].mass;e+=s.cards[i].energy;}return {mass:m,energy:e};}
function exact(s){for(var i=0;i<s.cards.length;i++)if(s.cards[i].bay<0)return false;for(var b=0;b<s.bays.length;b++){var t=totals(s,b),z=s.bays[b];if(t.mass!==z.mass||(z.energy!==null&&t.energy!==z.energy))return false;}return true;}
function activate(s,index){if(index<s.cards.length){s.selectedCard=index;s.focus=index;s.interactions++;return;}var b=index-s.cards.length;if(s.selectedCard>=0&&s.selectedCard<s.cards.length&&b>=0&&b<s.bays.length){var c=s.cards[s.selectedCard];if(c.bay!==b){c.bay=b;s.assignments++;s.interactions++;s.flash=18;s.validation=0;}s.focus=index;}}
function create(seed){var s={seed:seed>>>0,tick:0,stage:0,score:0,focus:0,selectedCard:0,assignments:0,interactions:0,validation:0,flash:0,phase:'play',moveLock:0,cards:[],bays:[]};board(s,0);return s;}
function step(s,input){input=input||{};if(s.phase==='incomplete'||s.phase==='finished')return s;s.tick++;if(s.flash>0)s.flash--;if(s.moveLock>0)s.moveLock--;
 if(s.phase==='dispatch'){if(s.dispatchTicks>0)s.dispatchTicks--;else {if(s.stage>=3){s.phase='finished';}else board(s,s.stage);}return s;}
 if(s.tick>=3000){s.phase='incomplete';return s;}
 if(input.pointer){var p=input.pointer;if(p.y>=.43&&p.y<=.92){var col=Math.min(2,Math.max(0,(p.x*3)|0)),row=p.y>=.70?1:0,ix=row*3+col;if(ix<s.cards.length)activate(s,ix);}else if(p.y>=.08&&p.y<=.39){var bx=Math.min(s.bays.length-1,Math.max(0,(p.x*s.bays.length)|0));activate(s,s.cards.length+bx);}return s;}
 var d=input.x||0;if(d&&s.moveLock===0){var count=s.cards.length+s.bays.length;s.focus=(s.focus+d+count)%count;s.moveLock=8;s.interactions++;}
 if(input.action)activate(s,s.focus);
 if(exact(s)){s.validation++;if(s.validation>=30){s.stage++;s.score+=100+Math.max(0,50-((s.tick/60)|0))+Math.max(0,18-s.assignments);s.phase='dispatch';s.dispatchTicks=48;s.validation=0;}}else s.validation=0;
 return s;}
function observe(s){return {tick:s.tick,score:s.score,progress:Math.min(100,((s.tick/3000)*100)|0),interactions:s.interactions,entities:s.cards.length+s.bays.length+3};}
function terminal(s){return s.phase==='incomplete'||(s.phase==='finished'&&s.tick>=900);}
window.CloudCargoCore={create:create,step:step,observe:observe,terminal:terminal};
}());
