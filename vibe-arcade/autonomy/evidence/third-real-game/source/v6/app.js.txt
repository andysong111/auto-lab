(() => {
  'use strict';
  const canvas = document.querySelector('[data-game-canvas]');
  const goal = document.querySelector('[data-ribbon-goal-progress]');
  const elapsed = document.querySelector('[data-ribbon-elapsed]');
  const score = document.querySelector('[data-game-score]');
  const best = document.querySelector('[data-ribbon-best]');
  const status = document.querySelector('[data-game-status]');
  const result = document.querySelector('[data-ribbon-result]');
  const detail = document.querySelector('[data-ribbon-detail]');
  let mediaQuery = null;
  let mediaListener = null;

  const setMotion = (event) => {
    // The event value is preferred because MediaQueryList implementations can
    // notify listeners before their public matches property is refreshed.
    const reduced = event && typeof event.matches === 'boolean'
      ? event.matches
      : !!(mediaQuery && mediaQuery.matches);
    document.documentElement.dataset.reducedMotion = reduced ? 'true' : 'false';
    if (globalThis.RibbonPresentation) globalThis.RibbonPresentation.setReducedMotion(reduced);
  };

  const draw = (ctx, snapshot, size) => {
    globalThis.RibbonShiftArt.draw(ctx, snapshot, size);
    const state = snapshot.state;
    const localBest = Number.isFinite(snapshot.best) ? snapshot.best : 0;
    goal.textContent = `${state.solved}/3`;
    elapsed.textContent = `ELAPSED ${Math.min(100, Math.floor(state.tick / 27))}%`;
    score.textContent = String(state.score);
    best.textContent = String(localBest);
    status.textContent = state.outcome === 'complete'
      ? 'All three mosaics are restored. The completed weave is on display.'
      : state.outcome === 'failure'
        ? `Time ended with ${state.solved}/3 mosaics restored.`
        : state.celebrate > 0
          ? `Mosaic ${state.solved}/3 restored — preparing the next weave.`
          : `Mosaic ${Math.min(3, state.stage + 1)} of 3 · choose a ribbon and shift it.`;
    const finished = globalThis.RibbonShiftCore.terminal(state);
    result.hidden = !finished;
    if (finished) {
      const complete = state.outcome === 'complete';
      result.querySelector('h2').textContent = complete ? 'MOSAICS RESTORED' : 'TIME UP';
      result.querySelector('[data-ribbon-result-count]').textContent = `${state.solved}/3`;
      detail.textContent = complete
        ? `All 3 mosaics restored · ${state.moves} shifts used · current score ${state.score} · device best ${localBest} local practice`
        : `${state.solved}/3 restored · ${state.moves} shifts used · current score ${state.score} · device best ${localBest} local practice`;
    }
  };

  const boot = (metadata) => {
    mediaQuery = globalThis.matchMedia('(prefers-reduced-motion: reduce)');
    mediaListener = (event) => setMotion(event);
    // Support both current browsers and lightweight MediaQueryList shims.
    if (typeof mediaQuery.addEventListener === 'function') mediaQuery.addEventListener('change', mediaListener);
    if (typeof mediaQuery.addListener === 'function') mediaQuery.addListener(mediaListener);
    mediaQuery.onchange = mediaListener;
    setMotion(null);
    globalThis.PlayJoltGameKit.create({ core: globalThis.RibbonShiftCore, draw, canvas, metadata });
    globalThis.addEventListener('pagehide', () => {
      if (!mediaQuery || !mediaListener) return;
      if (typeof mediaQuery.removeEventListener === 'function') mediaQuery.removeEventListener('change', mediaListener);
      if (typeof mediaQuery.removeListener === 'function') mediaQuery.removeListener(mediaListener);
      if (mediaQuery.onchange === mediaListener) mediaQuery.onchange = null;
    }, { once: true });
  };

  fetch('./manifest.json')
    .then((response) => {
      if (!response.ok) throw new Error('practice manifest unavailable');
      return response.json();
    })
    .then(boot)
    .catch((error) => {
      const errorNode = document.querySelector('[data-game-error]');
      errorNode.hidden = false;
      errorNode.textContent = `Could not start practice: ${error.message}`;
    });
})();
